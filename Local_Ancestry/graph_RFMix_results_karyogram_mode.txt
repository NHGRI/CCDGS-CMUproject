chrjamhap2 <- as.data.frame(sapply(chrjamhap2, as.numeric))

cbind(chrjamhap2, "mode"=apply(chrjamhap2, 1, function(x) names(which.max(table(x))))) -> cjamh2m

assign(paste("c",i,"jamh1m",sep=""), cjamh1m)
assign(paste("c",i,"jamh2m",sep=""), cjamh2m)
